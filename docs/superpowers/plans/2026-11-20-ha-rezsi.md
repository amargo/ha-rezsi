# ha-rezsi Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Home Assistant custom integration (HACS) that shows, per meter site, the electricity and gas cost of the current billing period at regulated (rezsicsökkentett) and market prices, from the user's own HA long-term statistics.

**Architecture:** Pure, HA-independent core (`tariffs.py` + `tariffs.yaml`, `calc.py`) ported from the web Rezsikalkulátor `calc()`. A thin HA layer (`statistics.py` → `coordinator.py` → sensor/button/date entities) fetches daily `change` statistics since the billing-period start (kept in a `Store`) and feeds the core hourly. One config entry = one meter site; entity ids derive from the entry title.

**Tech Stack:** Python 3.13, Home Assistant custom integration (config flow, DataUpdateCoordinator, recorder statistics), PyYAML (bundled with HA), pytest + pytest-homeassistant-custom-component, GitHub Actions (hassfest, HACS action).

```python
"""Dated tariff blocks: load, look up by day, apply user overrides. No HA imports."""
from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import date
from pathlib import Path
from typing import Any

import yaml

TARIFF_FILE = Path(__file__).with_name("tariffs.yaml")


@dataclass(frozen=True)
class Electricity:
    reduced: float
    above: float
    band_kwh: float
    market: float
    feed_in: float
    rhd_base_extra: float
    rhd_kwh_extra: float


@dataclass(frozen=True)
class Gas:
    reduced: float
    above: float
    band_m3: float
    market: float


@dataclass(frozen=True)
class Tariff:
    valid_from: date
    source: str
    electricity: Electricity
    gas: Gas
    vat: float


def _as_date(v: Any) -> date:
    return v if isinstance(v, date) else date.fromisoformat(str(v))


def load_tariffs(path: Path = TARIFF_FILE) -> list[Tariff]:
    """Load blocks sorted by valid_from; each block inherits unset keys."""
    raw = yaml.safe_load(path.read_text(encoding="utf-8")) or []
    if not raw:
        raise ValueError(f"{path.name}: no tariff blocks")
    out: list[Tariff] = []
    elec: dict[str, Any] = {}
    gas: dict[str, Any] = {}
    vat: Any = None
    for block in sorted(raw, key=lambda b: _as_date(b["valid_from"])):
        day = _as_date(block["valid_from"])
        elec = {**elec, **(block.get("electricity") or {})}
        gas = {**gas, **(block.get("gas") or {})}
        vat = block.get("vat", vat)
        try:
            out.append(
                Tariff(
                    valid_from=day,
                    source=str(block["source"]).strip(),
                    electricity=Electricity(**{k: float(v) for k, v in elec.items()}),
                    gas=Gas(**{k: float(v) for k, v in gas.items()}),
                    vat=float(vat),
                )
            )
        except (TypeError, KeyError) as err:
            raise ValueError(f"incomplete tariff block {day.isoformat()}: {err}") from err
    return out


def tariff_for(tariffs: list[Tariff], day: date) -> Tariff:
    """Return the last block with valid_from <= day."""
    found = [t for t in tariffs if t.valid_from <= day]
    if not found:
        raise ValueError(f"no tariff valid on {day.isoformat()}")
    return found[-1]


def apply_overrides(t: Tariff, overrides: dict[str, Any]) -> Tariff:
    """Overrides look like {"electricity.market": 150}; None/"" mean 'use default'."""
    parts: dict[str, dict[str, float]] = {"electricity": {}, "gas": {}}
    for key, value in overrides.items():
        if value in (None, ""):
            continue
        group, _, field = key.partition(".")
        if group in parts and field:
            parts[group][field] = float(value)
    return replace(
        t,
        electricity=replace(t.electricity, **parts["electricity"]),
        gas=replace(t.gas, **parts["gas"]),
    )
```

```python
"""Pure cost calculation, ported from the web Rezsikalkulátor calc(). No HA imports."""
from __future__ import annotations

from collections import defaultdict
from dataclasses import dataclass, field
from datetime import date
from typing import Callable

from .tariffs import Tariff


@dataclass(frozen=True)
class DayUsage:
    day: date
    imp: float = 0.0
    exp: float = 0.0
    gas: float = 0.0


@dataclass(frozen=True)
class CalcOptions:
    szaldo: bool = True
    elec_band_mult: float = 1.0
    gas_band_mult: float = 1.0
    rhd: bool = False


@dataclass
class Result:
    days: int
    imp_kwh: float
    exp_kwh: float
    billed_kwh: float
    band_kwh: float
    band_used_pct: float
    credit: float
    elec_reg: float
    elec_mkt: float
    rhd: float
    gas_m3: float
    gas_band_m3: float
    gas_reg: float
    gas_mkt: float
    tariff: Tariff
    monthly: dict[str, dict[str, float]] = field(default_factory=dict)


def tier(q: float, cap: float, lo: float, hi: float) -> float:
    """Banded price: up to cap at lo, above at hi."""
    return min(q, cap) * lo + max(0.0, q - cap) * hi


def compute(usage: list[DayUsage], tariff_for: Callable[[date], Tariff], opts: CalcOptions) -> Result:
    """Cost of the period covered by `usage` (one entry per day).

    Szaldó nets import and export over the whole period. Days are grouped by the
    tariff block valid on them; each group gets the band in proportion to its day
    count and the billed quantity in proportion to its import (or days if no import).
    """
    if not usage:
        raise ValueError("no usage data")
    groups: dict[date, list[DayUsage]] = defaultdict(list)
    blocks: dict[date, Tariff] = {}
    for u in usage:
        t = tariff_for(u.day)
        groups[t.valid_from].append(u)
        blocks[t.valid_from] = t

    n = len(usage)
    imp = sum(u.imp for u in usage)
    exp = sum(u.exp for u in usage)
    billed = max(0.0, imp - exp) if opts.szaldo else imp
    surplus = max(0.0, exp - imp) if opts.szaldo else exp

    band = credit = elec_reg = elec_mkt = rhd = 0.0
    gas_q = gas_band = gas_reg = gas_mkt = 0.0
    for key in sorted(groups):
        t, days = blocks[key], groups[key]
        d = len(days)
        e = t.electricity
        g_imp = sum(u.imp for u in days)
        g_exp = sum(u.exp for u in days)
        q = billed * (g_imp / imp if imp else d / n)
        b = e.band_kwh * opts.elec_band_mult * d / 365
        cr = surplus * (g_exp / exp if exp else d / n) * e.feed_in
        r = (e.rhd_base_extra * d / 365 + e.rhd_kwh_extra * q) * (1 + t.vat / 100) if opts.rhd else 0.0
        band += b
        credit += cr
        rhd += r
        elec_reg += tier(q, b, e.reduced, e.above) - cr
        elec_mkt += q * e.market - cr + r

        gq = sum(u.gas for u in days)
        gb = t.gas.band_m3 * opts.gas_band_mult * d / 365
        gas_q += gq
        gas_band += gb
        gas_reg += tier(gq, gb, t.gas.reduced, t.gas.above)
        gas_mkt += gq * t.gas.market

    monthly: dict[str, dict[str, float]] = {}
    for u in usage:
        m = monthly.setdefault(u.day.strftime("%Y-%m"), {"vetel_kwh": 0.0, "betaplalas_kwh": 0.0, "gaz_m3": 0.0})
        m["vetel_kwh"] += u.imp
        m["betaplalas_kwh"] += u.exp
        m["gaz_m3"] += u.gas
    monthly = {k: {kk: round(vv, 3) for kk, vv in v.items()} for k, v in monthly.items()}

    return Result(
        days=n,
        imp_kwh=imp,
        exp_kwh=exp,
        billed_kwh=billed,
        band_kwh=band,
        band_used_pct=billed / band * 100 if band else 0.0,
        credit=credit,
        elec_reg=elec_reg,
        elec_mkt=elec_mkt,
        rhd=rhd,
        gas_m3=gas_q,
        gas_band_m3=gas_band,
        gas_reg=gas_reg,
        gas_mkt=gas_mkt,
        tariff=blocks[max(blocks)],
        monthly=monthly,
    )
```

"""Daily consumption from recorder long-term statistics."""
from __future__ import annotations

from datetime import date, timedelta

from homeassistant.components.recorder import get_instance
from homeassistant.components.recorder.statistics import statistics_during_period
from homeassistant.core import HomeAssistant
from homeassistant.util import dt as dt_util


async def async_daily_changes(
    hass: HomeAssistant, stat_ids: dict[str, str | None], start: date, end: date
) -> dict[str, dict[date, float]]:
    """Return {key: {local day: change}} for start..end (inclusive)."""
    wanted = {k: v for k, v in stat_ids.items() if v}
    if not wanted:
        return {}
    start_dt = dt_util.start_of_local_day(start)
    end_dt = dt_util.start_of_local_day(end + timedelta(days=1))
    rows = await get_instance(hass).async_add_executor_job(
        statistics_during_period,
        hass,
        start_dt,
        end_dt,
        set(wanted.values()),
        "day",
        None,
        {"change"},
    )
    out: dict[str, dict[date, float]] = {}
    for key, sid in wanted.items():
        out[key] = {
            dt_util.as_local(dt_util.utc_from_timestamp(r["start"])).date(): float(r.get("change") or 0.0)
            for r in rows.get(sid, [])
        }
    return out

"""Billing-period state per config entry."""
from __future__ import annotations

from datetime import date
from typing import Any

from homeassistant.core import HomeAssistant
from homeassistant.helpers.storage import Store

from .const import DOMAIN


class PeriodStore:
    """Period start and the last closed period, kept in .storage."""

    def __init__(self, hass: HomeAssistant, entry_id: str) -> None:
        self._store: Store[dict[str, Any]] = Store(hass, 1, f"{DOMAIN}.{entry_id}")
        self.period_start: date = date.today()
        self.previous: dict[str, Any] | None = None

    async def async_load(self, default_start: date) -> None:
        data = await self._store.async_load() or {}
        start = data.get("period_start")
        self.period_start = date.fromisoformat(start) if start else default_start
        self.previous = data.get("previous")

    async def async_set_start(self, day: date) -> None:
        self.period_start = day
        await self._save()

    async def async_close_period(self, today: date, summary: dict[str, Any]) -> None:
        self.previous = {**summary, "idoszak_vege": today.isoformat()}
        self.period_start = today
        await self._save()

    async def async_remove(self) -> None:
        await self._store.async_remove()

    async def _save(self) -> None:
        await self._store.async_save(
            {"period_start": self.period_start.isoformat(), "previous": self.previous}
        )
```

"""Hourly recomputation for one meter site."""
from __future__ import annotations

import logging
from datetime import date, timedelta
from typing import Any

from homeassistant.config_entries import ConfigEntry
from homeassistant.core import HomeAssistant
from homeassistant.helpers.update_coordinator import DataUpdateCoordinator, UpdateFailed
from homeassistant.util import dt as dt_util

from .calc import CalcOptions, DayUsage, Result, compute
from .const import (
    CONF_ELEC_MULT, CONF_EXPORT, CONF_GAS, CONF_GAS_MULT, CONF_IMPORT, CONF_OVERRIDES, CONF_RHD, CONF_SZALDO,
)
from .statistics import async_daily_changes
from .store import PeriodStore
from .tariffs import Tariff, apply_overrides, tariff_for

_LOGGER = logging.getLogger(__name__)


class RezsiCoordinator(DataUpdateCoordinator[Result]):
    """Fetch daily statistics since the period start and compute costs."""

    def __init__(self, hass: HomeAssistant, entry: ConfigEntry, store: PeriodStore, tariffs: list[Tariff]) -> None:
        super().__init__(hass, _LOGGER, config_entry=entry, name=entry.title, update_interval=timedelta(hours=1))
        self.store = store
        self.tariffs = tariffs
        self.gas_ok = False
        self.first_day: date | None = None

    @property
    def cfg(self) -> dict[str, Any]:
        return {**self.config_entry.data, **self.config_entry.options}

    @property
    def has_gas(self) -> bool:
        return bool(self.cfg.get(CONF_GAS))

    async def _async_update_data(self) -> Result:
        cfg = self.cfg
        start = self.store.period_start
        today = dt_util.now().date()
        ids = {"imp": cfg.get(CONF_IMPORT), "exp": cfg.get(CONF_EXPORT), "gas": cfg.get(CONF_GAS)}
        changes = await async_daily_changes(self.hass, ids, start, today)
        imp = changes.get("imp") or {}
        if not imp:
            raise UpdateFailed(f"no statistics for {cfg.get(CONF_IMPORT)}")
        exp = changes.get("exp") or {}
        gas = changes.get("gas") or {}
        self.gas_ok = bool(gas)
        self.first_day = min(imp)
        usage = [
            DayUsage(d, imp.get(d, 0.0), exp.get(d, 0.0), gas.get(d, 0.0))
            for d in (start + timedelta(days=n) for n in range((today - start).days + 1))
        ]
        overrides = cfg.get(CONF_OVERRIDES) or {}
        opts = CalcOptions(
            szaldo=bool(cfg.get(CONF_SZALDO, True)),
            elec_band_mult=float(cfg.get(CONF_ELEC_MULT, 1)),
            gas_band_mult=float(cfg.get(CONF_GAS_MULT, 1)),
            rhd=bool(cfg.get(CONF_RHD, False)),
        )
        try:
            return compute(usage, lambda d: apply_overrides(tariff_for(self.tariffs, d), overrides), opts)
        except ValueError as err:
            raise UpdateFailed(str(err)) from err

    def totals(self) -> tuple[float, float]:
        """(regulated, market) for electricity plus gas when available."""
        r = self.data
        gas = self.has_gas and self.gas_ok
        return (r.elec_reg + (r.gas_reg if gas else 0.0), r.elec_mkt + (r.gas_mkt if gas else 0.0))

    def summary(self) -> dict[str, Any]:
        """Snapshot stored when the period is closed."""
        r = self.data
        reg, mkt = self.totals()
        return {
            "idoszak_kezdete": self.store.period_start.isoformat(),
            "eltelt_napok": r.days,
            "villany_rezsiaron": round(r.elec_reg),
            "villany_piaci_aron": round(r.elec_mkt),
            "gaz_rezsiaron": round(r.gas_reg) if self.gas_ok else None,
            "gaz_piaci_aron": round(r.gas_mkt) if self.gas_ok else None,
            "osszesen_rezsiaron": round(reg),
            "osszesen_piaci_aron": round(mkt),
            "elszamolt_kwh": round(r.billed_kwh, 1),
            "gaz_m3": round(r.gas_m3, 1),
        }

    def attributes(self) -> dict[str, Any]:
        r = self.data
        return {
            "idoszak_kezdete": self.store.period_start.isoformat(),
            "eltelt_napok": r.days,
            "vetel_kwh": round(r.imp_kwh, 1),
            "betaplalas_kwh": round(r.exp_kwh, 1),
            "elszamolt_kwh": round(r.billed_kwh, 1),
            "gaz_m3": round(r.gas_m3, 1),
            "savhatar_aranyos": round(r.band_kwh, 1),
            "savhatar_felhasznalt_pct": round(r.band_used_pct, 1),
            "gaz_savhatar_aranyos": round(r.gas_band_m3, 1),
            "tarifa_valid_from": r.tariff.valid_from.isoformat(),
            "tarifa_forras": r.tariff.source,
            "havi_bontas": r.monthly,
            "elozo_idoszak": self.store.previous,
            "adat_hianyos": bool(self.first_day and self.first_day > self.store.period_start),
            "elso_adat_napja": self.first_day.isoformat() if self.first_day else None,
        }
```

async def async_setup_entry(hass: HomeAssistant, entry: RezsiConfigEntry) -> bool:
    tariffs = await hass.async_add_executor_job(load_tariffs)
    store = PeriodStore(hass, entry.entry_id)
    start = entry.data.get(CONF_PERIOD_START)
    default = date.fromisoformat(start) if start else dt_util.now().date() - timedelta(days=365)

    await hass.config_entries.async_forward_entry_setups(entry, PLATFORMS)
    entry.async_on_unload(entry.add_update_listener(_async_reload))

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass
from typing import Any

from homeassistant.components.sensor import SensorDeviceClass, SensorEntity
from homeassistant.core import HomeAssistant
from homeassistant.helpers.entity_platform import AddEntitiesCallback

from . import RezsiConfigEntry
from .coordinator import RezsiCoordinator
from .entity import RezsiEntity


@dataclass(frozen=True)
class Desc:
    key: str
    name: str
    value: Callable[[RezsiCoordinator], float]
    gas: bool = False


def _diff(c: RezsiCoordinator) -> float:
    reg, mkt = c.totals()
    return mkt - reg


DESCS = (
    Desc("villany_rezsiaron", "Villany rezsiáron", lambda c: c.data.elec_reg),
    Desc("villany_piaci_aron", "Villany piaci áron", lambda c: c.data.elec_mkt),
    Desc("gaz_rezsiaron", "Gáz rezsiáron", lambda c: c.data.gas_reg, gas=True),
    Desc("gaz_piaci_aron", "Gáz piaci áron", lambda c: c.data.gas_mkt, gas=True),
    Desc("osszesen_rezsiaron", "Összesen rezsiáron", lambda c: c.totals()[0]),
    Desc("osszesen_piaci_aron", "Összesen piaci áron", lambda c: c.totals()[1]),
    Desc("kulonbseg", "Különbség", _diff),
    Desc("eves_becsles_piaci", "Éves becslés piaci", lambda c: c.totals()[1] * 365 / max(c.data.days, 1)),
)


async def async_setup_entry(hass: HomeAssistant, entry: RezsiConfigEntry, add: AddEntitiesCallback) -> None:
    c = entry.runtime_data
    add(RezsiSensor(c, d) for d in DESCS if not d.gas or c.has_gas)


class RezsiSensor(RezsiEntity, SensorEntity):
    _attr_device_class = SensorDeviceClass.MONETARY
    _attr_native_unit_of_measurement = "Ft"
    _attr_suggested_display_precision = 0

    def __init__(self, coordinator: RezsiCoordinator, desc: Desc) -> None:
        super().__init__(coordinator, desc.key, desc.name)
        self._desc = desc

    @property
    def available(self) -> bool:
        return super().available and (not self._desc.gas or self.coordinator.gas_ok)

    @property
    def native_value(self) -> float:
        return round(self._desc.value(self.coordinator), 2)

    @property
    def extra_state_attributes(self) -> dict[str, Any]:
        attrs = self.coordinator.attributes()
        if self._desc.key == "kulonbseg":
            reg, _ = self.coordinator.totals()
            attrs["kulonbseg_pct"] = round(_diff(self.coordinator) / reg * 100, 1) if reg else None
        return attrs

Mennyit fizetsz most rezsicsökkentett áron, és mennyit fizetnél piaci áron? Az integráció a
